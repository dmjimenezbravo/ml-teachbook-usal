\begin{frame}{¿Qué es el aprendizaje supervisado?}
  \begin{itemize}
    \item Tenemos \textbf{datos} (características de entrada, $X$) y \textbf{etiquetas} (respuestas correctas, $y$)
    \item El objetivo es aprender la relación $X \rightarrow y$
    \item Es como aprender con un ``supervisor'' que corrige tus respuestas durante el entrenamiento
  \end{itemize}
  \vspace{1em}
  \begin{columns}
    \begin{column}{0.5\textwidth}

        \item Temperatura de mañana
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Clasificación} (categoría)
      \begin{itemize}

        \item ¿Email es \textit{spam}?
        \item ¿Imagen es gato, perro o pájaro?
      \end{itemize}
    \end{column}
  \end{columns}

\section{2.1 Regresión}

\begin{frame}{Regresión lineal}
  \begin{itemize}
    \item \textbf{Simple}: un único predictor, $Y \approx \beta_0 + \beta_1 X$
    \item \textbf{Múltiple}: un coeficiente por cada característica, $Y = \beta_0 + \beta_1X_1 + \dots + \beta_pX_p + \epsilon$
    \item $\beta_0$ (intercepto) y $\beta_1 \dots \beta_p$ (pesos) son lo que el modelo aprende de los datos
    \item En $p$ dimensiones, el modelo representa un \textbf{hiperplano}
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.5\textwidth]{../book/_static/generated/figures/es/linear_regression_fit.png}
  \end{center}
\end{frame}

\begin{frame}{Ejemplo real: dataset \textit{Advertising}}
  \begin{itemize}
    \item Predice las ventas (\textit{Sales}) a partir de la inversión en TV, radio y prensa
    \item La recta minimiza la suma de residuos al cuadrado (OLS)
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.48\textwidth]{../book/_static/book_figures/islr_fig3_1_advertising.png}
  \end{center}
  \fuente{James, Witten, Hastie \& Tibshirani (2013), \textit{An Introduction to Statistical Learning}, Fig. 3.1. Libro de libre distribución (statlearning.com).}
\end{frame}

\begin{frame}{Mínimos cuadrados ordinarios (OLS)}
  \begin{itemize}
    \item Un \textbf{residual} $e_i = y_i - \hat{y}_i$ es la diferencia entre el valor real y el predicho
    \item OLS busca los pesos que \textbf{minimizan} la suma de los residuos al cuadrado: $RSS = \sum_{i=1}^{n}(y_i-\hat{y}_i)^2$
    \item Tiene una \textbf{solución analítica exacta}: $\hat{\beta} = (X^TX)^{-1}X^Ty$
    \item Interpretación geométrica: $\hat{y}$ es la proyección ortogonal de $y$ sobre el espacio definido por $X$
  \end{itemize}
  \ejemplo{con solo 3 casas (m\textsuperscript{2} y precio), OLS encuentra en un único paso la recta que mejor resume esa relación --- sin necesidad de ensayo y error.}
\end{frame}

\begin{frame}{Descenso de gradiente}
  \begin{itemize}
    \item Cuando los datos son masivos, resolver la ecuación exacta es demasiado costoso
    \item Algoritmo \textbf{iterativo}: parte de pesos aleatorios y los ajusta poco a poco en la dirección que más reduce el error
    \item $\theta \leftarrow \theta - \alpha \nabla J(\theta)$, donde $\alpha$ es la \textbf{tasa de aprendizaje}
    \item Variantes: \textit{Batch} (todo el dataset), \textit{Stochastic} (una muestra), \textit{Mini-batch} (un grupo pequeño)
  \end{itemize}
  \ejemplo{\footnotesize es como bajar una montaña con niebla: en cada paso miras la pendiente bajo tus pies y avanzas hacia donde más desciende, hasta llegar al valle.}
  \begin{center}
    \includegraphics[width=0.32\textwidth]{../book/_static/generated/figures/es/gradient_descent.png}
  \end{center}
\end{frame}

\begin{frame}{Regresión polinomial y el riesgo de sobreajuste}
  \begin{itemize}
    \item Se añaden potencias de los predictores ($X^2$, $X^3$...) para capturar curvas
    \item Sigue siendo un modelo \textbf{lineal en sus parámetros} --- se puede seguir entrenando con OLS
    \item A mayor grado, mayor riesgo de memorizar el ruido (\textit{overfitting})
  \end{itemize}
\end{frame}

\begin{frame}{Regularización: \textit{Ridge} y \textit{Lasso}}
  \begin{itemize}
    \item Penalizan los pesos grandes en la función de coste, forzando modelos más simples
    \item \textbf{\textit{Ridge}} ($L_2$): penaliza $\sum \beta_j^2$ --- encoge los coeficientes sin anularlos
    \item \textbf{\textit{Lasso}} ($L_1$): penaliza $\sum |\beta_j|$ --- puede llevar coeficientes exactamente a cero (selección de variables)
    \item \textbf{\textit{Elastic Net}}: combina ambas penalizaciones
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.55\textwidth]{../book/_static/generated/figures/es/regularization_paths.png}
  \end{center}
\end{frame}

\begin{frame}{Ridge en datos reales: dataset \textit{Credit}}
  \begin{itemize}
    \item Al aumentar $\lambda$, los coeficientes se encogen hacia cero
    \item Reduce la varianza a cambio de un pequeño aumento del sesgo
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.6\textwidth]{../book/_static/book_figures/islr_fig6_4_ridge_credit.png}
  \end{center}
  \fuente{James, Witten, Hastie \& Tibshirani (2013), \textit{An Introduction to Statistical Learning}, Fig. 6.4.}
\end{frame}

\begin{frame}{Validación y diagnóstico en regresión}
  \begin{itemize}
    \item \textbf{MSE / RMSE}: error cuadrático medio (y su raíz); penaliza fuerte los errores grandes
    \item \textbf{MAE}: error absoluto medio; más robusto ante valores atípicos
    \item \textbf{$R^2$}: proporción de varianza explicada por el modelo (1 = perfecto, 0 = tan bueno como predecir la media)
    \item \textbf{Curvas de validación}: graficar error de entrenamiento y validación frente a la complejidad revela si hay sesgo alto (ambos errores altos) o varianza alta (brecha entre ambos)
  \end{itemize}
\end{frame}

\begin{frame}{Resumen: 2.1 Regresión}
  \begin{itemize}
    \item \textbf{Regresión lineal}: modelo simple pero muy potente
    \item \textbf{OLS}: solución exacta; \textbf{descenso de gradiente}: método general y escalable
    \item \textbf{Regresión polinomial}: captura relaciones no lineales, a costa de más riesgo de \textit{overfitting}
    \item \textbf{Regularización}: \textit{Ridge} encoge, \textit{Lasso} selecciona variables
  \end{itemize}
\end{frame}

\section{2.2 Clasificación}

\begin{frame}{Clasificación: predecir categorías}
  \begin{itemize}
    \item El objetivo es predecir una \textbf{categoría}, no un número continuo
    \item Ejemplos: \textit{spam} vs. no-\textit{spam}, enfermo vs. sano, gato vs. perro vs. pájaro
    \item La \textbf{regresión logística} empieza estimando la \textbf{probabilidad} de pertenecer a una clase
  \end{itemize}
\end{frame}

\begin{frame}{La función sigmoide}
  \begin{itemize}
    \item Transforma cualquier valor real en una probabilidad entre 0 y 1
    \item $\sigma(t) = \dfrac{1}{1+e^{-t}}$, donde $t$ es el \textit{logit} (combinación lineal de las \textit{features})
    \item Se entrena minimizando la \textbf{entropía cruzada} (\textit{log loss}) con descenso de gradiente
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.42\textwidth]{../book/_static/generated/figures/es/sigmoid_function.png}
  \end{center}
\end{frame}

\begin{frame}{¿Por qué no usar regresión lineal?}
  \begin{itemize}
    \item La regresión lineal puede predecir probabilidades $<0$ o $>1$ (sin sentido)
    \item La logística usa la sigmoide para mantener la salida siempre en $[0,1]$
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.62\textwidth]{../book/_static/book_figures/islr_fig4_2_default_logistic.png}
  \end{center}
  \fuente{James, Witten, Hastie \& Tibshirani (2013), \textit{An Introduction to Statistical Learning}, Fig. 4.2.}
\end{frame}

\begin{frame}{Árboles de decisión}
  \begin{columns}
    \begin{column}{0.45\textwidth}
      \begin{itemize}
        \item Dividen el espacio de \textit{features} con cortes sucesivos ortogonales, como un diagrama de flujo de preguntas ``sí/no''
        \item Algoritmo \textbf{CART}: cada división busca maximizar la pureza de los nodos hijos (índice Gini o entropía)
        \item Fáciles de interpretar, pero propensos a \textit{overfitting} sin \textbf{poda}
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\textwidth]{../book/_static/generated/figures/es/decision_tree_partition.png}
    \end{column}
  \end{columns}
  \ejemplo{``¿el email contiene la palabra `gratis'\,? $\rightarrow$ sí $\rightarrow$ ¿tiene más de 3 enlaces? $\rightarrow$ sí $\rightarrow$ \textit{spam}''.}
\end{frame}

\begin{frame}{Bosques aleatorios (\textit{Random Forests})}
  \begin{itemize}
    \item Combinan \textbf{muchos árboles} para reducir la varianza y mejorar la robustez
    \item \textbf{\textit{Bagging}}: cada árbol se entrena con una muestra \textit{bootstrap} distinta (con reemplazo)
    \item \textbf{Decorrelación}: en cada división solo se considera un subconjunto aleatorio de \textit{features}
  \end{itemize}
  \footnotesize
  Varias muestras \textit{bootstrap}, un árbol distinto por cada una, y las predicciones se combinan por votación mayoritaria o promedio.
  \begin{center}
    \includegraphics[width=0.5\textwidth]{../book/_static/generated/diagrams_pdf/es/01_aprendizaje_supervisado_02_clasificacion_01.png}
  \end{center}
\end{frame}

\begin{frame}{SVM: margen máximo}
  \begin{columns}
    \begin{column}{0.45\textwidth}
      \begin{itemize}
        \item Busca el \textbf{hiperplano} que separa las clases con el \textbf{margen máximo}
        \item Los \textbf{vectores de soporte} son los únicos puntos que determinan la frontera
        \item El \textit{kernel trick} permite fronteras no lineales, mapeando los datos a un espacio de más dimensiones
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \centering
      \includegraphics[width=\textwidth]{../book/_static/generated/figures/es/svm_margin.png}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{\textit{Voting} y \textit{stacking}}
  \begin{itemize}
    \item Combinan \textbf{varios modelos} para lograr una predicción mejor que cualquiera por separado
    \item \textbf{\textit{Voting}}: voto mayoritario o promedio de probabilidades, con una regla fija
    \item \textbf{\textit{Stacking}}: un meta-modelo \textbf{aprende} cómo combinar las salidas de los modelos base
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.68\textwidth]{../book/_static/generated/diagrams_pdf/es/01_aprendizaje_supervisado_02_clasificacion_03.png}
  \end{center}
\end{frame}

\begin{frame}{\textit{Boosting}}
  \begin{itemize}
    \item A diferencia del entrenamiento en paralelo de los bosques aleatorios, el \textit{boosting} entrena modelos de forma \textbf{secuencial}
    \item Cada nuevo modelo intenta corregir los errores del anterior, dando más peso a las instancias mal clasificadas
    \item Ejemplos modernos: \textit{XGBoost}, \textit{Gradient Boosting}
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.85\textwidth]{../book/_static/generated/diagrams_pdf/es/01_aprendizaje_supervisado_02_clasificacion_02.png}
  \end{center}
\end{frame}

\begin{frame}{Redes neuronales}
  \begin{itemize}
    \item Capas de neuronas conectadas por pesos ajustables
    \item Cada neurona aplica una función de activación no lineal: \textbf{ReLU} en capas ocultas, \textbf{Softmax} en la capa de salida
    \item \textbf{Retropropagación} (\textit{backpropagation}): propaga el error desde la salida hacia atrás para ajustar todos los pesos
  \end{itemize}
  \begin{center}
    \includegraphics[width=0.42\textwidth]{../book/_static/generated/figures/es/nn_architecture.png}
  \end{center}
\end{frame}

\begin{frame}{Evaluación de clasificadores}
  \begin{itemize}
    \item \textbf{Exactitud} (\textit{accuracy}): puede engañar si las clases están desbalanceadas
    \item \textbf{Precisión} y \textbf{\textit{recall}}: qué tan fiables son los positivos vs. cuántos positivos reales se encuentran
    \item \textbf{F1}: equilibrio entre ambas
    \item \textbf{ROC-AUC}: compara modelos sin depender de un umbral fijo
  \end{itemize}
  \ejemplo{\footnotesize si el 99\% de las transacciones no son fraude, un modelo que siempre diga ``no es fraude'' tendría 99\% de exactitud sin haber aprendido nada útil.}
  \begin{center}
    \includegraphics[width=0.26\textwidth]{../book/_static/generated/figures/es/roc_curve.png}
  \end{center}
\end{frame}

\begin{frame}{Estrategias de entrenamiento y diagnóstico}
  \begin{itemize}
    \item \textbf{Superar una línea de base}: antes de complicarse, comprobar que se le gana a un clasificador trivial (aleatorio)
    \item \textbf{Curvas de aprendizaje}: pérdida y exactitud en entrenamiento/validación revelan el punto exacto de sobreajuste
    \item \textbf{\textit{Early stopping}}: detener el entrenamiento cuando la validación deja de mejorar
    \item \textbf{Análisis de errores}: revisar a mano los casos donde el modelo falla
  \end{itemize}
\end{frame}

\begin{frame}{Resumen: 2.2 Clasificación}
  \begin{itemize}
    \item \textbf{Regresión logística}: extiende la regresión lineal a probabilidades
    \item \textbf{Árboles}: interpretables, propensos a \textit{overfitting}; \textbf{\textit{Random Forest}}: lo corrige combinando muchos
    \item \textbf{SVM}: separación óptima, con \textit{kernel trick} para fronteras complejas
    \item \textbf{Ensambles} (\textit{voting}, \textit{stacking}, \textit{boosting}): combinan modelos para mejorar
    \item \textbf{Redes neuronales}: potentes pero más complejas de interpretar
  \end{itemize}
\end{frame}
